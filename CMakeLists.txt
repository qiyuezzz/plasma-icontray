cmake_minimum_required(VERSION 3.24)
project(IconTray VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
find_package(ECM 6.30 REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)
include(ECMQtDeclareLoggingCategory)
include(ECMQmlModule)
find_package(Qt6 6.11 REQUIRED COMPONENTS DBus Gui Quick Widgets)
find_package(Qt6GuiPrivate ${Qt6_VERSION} REQUIRED NO_MODULE)
foreach(component Config CoreAddons IconThemes I18n KIO JobWidgets Service ItemModels WindowSystem XmlGui StatusNotifierItem)
    find_package(KF6${component} 6.30 REQUIRED)
endforeach()
find_package(Plasma 6.8 REQUIRED)
find_package(PlasmaQuick 6.8 REQUIRED)
# This source snapshot uses the 6.8 applet/containment API.
if(Plasma_VERSION VERSION_GREATER_EQUAL 6.9)
    message(WARNING "IconTray is tested with Plasma 6.8; verify compatibility with ${Plasma_VERSION}.")
endif()
ki18n_install(po)
include(CTest)
add_subdirectory(vendor/libdbusmenuqt)
add_subdirectory(src)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_subdirectory(tests)
endif()

# Keep an uninstaller alongside the installation, independent of this checkout.
install(PROGRAMS tools/icontray-uninstall DESTINATION bin)
install(FILES tools/manage.py DESTINATION share/icontray)
